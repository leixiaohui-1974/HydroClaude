# 示例1：河道水流计算
from backend.core.structures.channel import Channel, ChannelType, CrossSectionShape

canal = Channel(
    name="Main-Canal",
    channel_type=ChannelType.CANAL,
    length=5000.0,
    shape=CrossSectionShape.RECTANGULAR,
    width=10.0,
    depth=3.0,
    slope=0.0005,
    manning_n=0.020
)

# 计算正常水深
Q = 30.0  # m³/s
h_normal = canal.compute_normal_depth(Q)
h_critical = canal.compute_critical_depth(Q)
print(f"正常水深: {h_normal:.2f}m")
print(f"临界水深: {h_critical:.2f}m")

---

# 示例2：水库调度
from backend.core.structures.reservoir import Reservoir

reservoir = Reservoir(
    name="Test-Reservoir",
    dead_level=100.0,
    normal_level=150.0,
    flood_limit_level=145.0,
    design_flood_level=155.0,
    check_flood_level=160.0,
    elevation=[100, 120, 140, 150, 160],
    area=[1e6, 2e6, 3e6, 4e6, 5e6],
    initial_level=145.0
)

# 模拟洪水过程
Q_in = 100.0   # 入流 m³/s
Q_out = 50.0   # 出流 m³/s
dt = 3600.0    # 时间步长 1小时

new_level, new_volume = reservoir.update(Q_in, Q_out, dt)
print(f"新水位: {new_level:.2f}m")
print(f"状态: {reservoir.check_level_status()}")

---

# 示例3：水轮机发电
from backend.core.structures.turbine import Turbine, TurbineType

turbine = Turbine(
    name="Main-Turbine",
    turbine_type=TurbineType.FRANCIS,
    rated_power=50.0,  # MW
    rated_speed=150.0  # rpm
)

H = 100.0  # 水头 m
Q = 50.0   # 流量 m³/s

efficiency = turbine.compute_efficiency(H, Q)
power = turbine.compute_power(H, Q)
print(f"效率: {efficiency*100:.1f}%")
print(f"功率: {power:.2f}MW")
