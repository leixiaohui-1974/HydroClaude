### 格式1: YAML格式 (ModelConfig)

**用途**: 适合人工编写和维护的配置文件  
**位置**: `/workspace/modeling/config.py`

#### 完整示例

```yaml
# model_config.yaml
canal:
  length: 1000.0        # 渠道长度 (m)
  width: 10.0           # 渠道宽度 (m)
  slope: 0.001          # 底坡
  manning_n: 0.025      # 曼宁糙率系数
  gravity: 9.81         # 重力加速度 (m/s²)

structures:
  - type: sluice_gate
    position: 500.0
    width: 10.0
    opening: 2.0
  - type: weir
    position: 750.0
    width: 10.0
    crest_height: 0.5

simulation:
  n_cells: 100          # 网格数量
  cfl: 0.5              # CFL数
  dt_max: 0.1           # 最大时间步 (s)
  t_end: 100.0          # 结束时间 (s)
  output_interval: 1.0  # 输出间隔 (s)
  order: 2              # 空间精度阶数
  use_numba: true       # 启用加速

initial_conditions:
  type: uniform         # 类型: uniform, dam_break, custom
  h: 5.0                # 初始水深 (m)
  Q: 10.0               # 初始流量 (m³/s)

boundary_conditions:
  upstream:
    type: Q             # 类型: Q (流量), h (水深), wall (反射)
    value: 10.0
  downstream:
    type: h
    value: 5.0

output:
  directory: ./results
  save_csv: true
  save_plots: true
```

#### Pydantic模型定义

```python
from pydantic import BaseModel, Field
from typing import Literal, Optional

class UpstreamBoundaryConfig(BaseModel):
    type: Literal['h', 'Q', 'wall'] = 'h'
    value: Optional[float] = None

class DownstreamBoundaryConfig(BaseModel):
    type: Literal['h', 'Q', 'wall'] = 'h'
    value: Optional[float] = None

class BoundaryConditionConfig(BaseModel):
    upstream: UpstreamBoundaryConfig
    downstream: DownstreamBoundaryConfig

class InitialConditionConfig(BaseModel):
    type: Literal['uniform', 'dam_break', 'custom'] = 'uniform'
    # Uniform flow
    h: Optional[float] = 5.0
    Q: Optional[float] = 0.0
    # Dam break
    dam_position: Optional[float] = None
    h_left: Optional[float] = None
    h_right: Optional[float] = None
    Q_left: Optional[float] = 0.0
    Q_right: Optional[float] = 0.0

class SimulationConfig(BaseModel):
    # Geometry (REQUIRED)
    width: float = Field(..., gt=0, le=1000)
    length: float = Field(..., gt=0, le=100000)
    n_cells: int = Field(..., ge=10, le=10000)
    
    # Physical parameters
    manning_n: float = Field(0.025, ge=0.001, le=0.1)
    slope: float = Field(0.001, ge=0, le=0.1)
    
    # Numerical parameters
    cfl: float = Field(0.5, gt=0, le=1.0)
    order: int = Field(2, ge=1, le=2)
    use_numba: bool = Field(True)
    
    # Time parameters
    t_end: float = Field(10.0, gt=0)
    dt_max: float = Field(0.1, gt=0)
    output_interval: float = Field(0.5, gt=0)
    
    # Conditions (REQUIRED)
    initial_conditions: InitialConditionConfig
    boundary_conditions: BoundaryConditionConfig
```

### SimulationResult (dataclass)

**用途**: 核心计算引擎返回的结果  
**位置**: `/workspace/web/backend/core/hydraulic_engine.py`

#### 数据结构

```python
from dataclasses import dataclass
from typing import List, Dict, Optional

@dataclass
class SimulationResult:
    """仿真结果数据类"""
    task_id: str                      # 任务ID
    status: str                       # 状态: completed, failed
    time: List[float]                 # 时间序列 [t1, t2, ..., tn]
    x: List[float]                    # 空间坐标 [x1, x2, ..., xm]
    h: List[List[float]]              # 水深 [时间][空间]
    Q: List[List[float]]              # 流量 [时间][空间]
    V: List[List[float]]              # 流速 [时间][空间]
    metrics: Dict[str, float]         # 性能指标
    duration: float                   # 计算用时 (秒)
    timestamp: str                    # 完成时间戳
    error: Optional[str] = None       # 错误信息
```

```python
class SimulationMetrics(BaseModel):
    """性能指标"""
    mass_conservation_error: float    # 质量守恒误差
    max_depth: float                  # 最大水深 (m)

class SimulationResultResponse(BaseModel):
    """API结果响应"""
    task_id: str
    status: str
    duration: float
    timestamp: datetime
```

## 🔄 格式转换工具

### YAML ↔ JSON 转换

创建转换工具文件：`/workspace/web/backend/utils/config_converter.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
配置格式转换工具
"""

import yaml
import json
from typing import Dict, Any
from pathlib import Path
from modeling.config import ModelConfig
from api_gateway.models.simulation import SimulationConfig

class ConfigConverter:
    """配置格式转换器"""
    
    @staticmethod
    def yaml_to_json_config(yaml_file: str) -> Dict[str, Any]:
        """
        YAML配置文件 -> JSON配置字典
        
        Args:
            yaml_file: YAML配置文件路径
            
        Returns:
            JSON格式的配置字典
        """
        # 加载YAML配置
        model_config = ModelConfig(yaml_file)
        
        # 转换为JSON格式
        canal = model_config.config['canal']
        simulation = model_config.config['simulation']
        initial = model_config.config.get('initial_conditions', {})
        boundary = model_config.config.get('boundary_conditions', {})
        
        json_config = {
            "width": canal['width'],
            "length": canal['length'],
            "n_cells": simulation['n_cells'],
            "manning_n": canal['manning_n'],
            "slope": canal['slope'],
            "cfl": simulation['cfl'],
            "order": simulation.get('order', 2),
            "use_numba": simulation.get('use_numba', True),
            "t_end": simulation['t_end'],
            "dt_max": simulation['dt_max'],
            "output_interval": simulation['output_interval'],
            "initial_conditions": initial,
            "boundary_conditions": boundary
        }
        
        return json_config
    
    @staticmethod
    def json_to_yaml_config(json_config: Dict[str, Any], output_file: str):
        """
        JSON配置字典 -> YAML配置文件
        
        Args:
            json_config: JSON格式的配置字典
            output_file: 输出YAML文件路径
        """
        # 转换为YAML格式
        yaml_config = {
            "canal": {
                "length": json_config["length"],
                "width": json_config["width"],
                "slope": json_config["slope"],
                "manning_n": json_config["manning_n"],
                "gravity": 9.81
            },
            "simulation": {
                "n_cells": json_config["n_cells"],
                "cfl": json_config["cfl"],
                "dt_max": json_config["dt_max"],
                "t_end": json_config["t_end"],
                "output_interval": json_config["output_interval"],
                "order": json_config.get("order", 2),
                "use_numba": json_config.get("use_numba", True)
            },
            "initial_conditions": json_config["initial_conditions"],
            "boundary_conditions": json_config["boundary_conditions"]
        }
        
        # 保存为YAML文件
        with open(output_file, 'w', encoding='utf-8') as f:
            yaml.dump(yaml_config, f, default_flow_style=False, allow_unicode=True)
    
    @staticmethod
    def validate_json_config(json_config: Dict[str, Any]) -> bool:
        """
        验证JSON配置是否有效
        
        Args:
            json_config: JSON配置字典
            
        Returns:
            True if valid, raises ValidationError otherwise
        """
        try:
            SimulationConfig(**json_config)
            return True
        except Exception as e:
            raise ValueError(f"配置验证失败: {e}")


# 使用示例
if __name__ == "__main__":
    converter = ConfigConverter()
    
    # YAML -> JSON
    json_config = converter.yaml_to_json_config("model_config.yaml")
    print("JSON配置:")
    print(json.dumps(json_config, indent=2, ensure_ascii=False))
    
    # 验证
    is_valid = converter.validate_json_config(json_config)
    print(f"\n配置有效: {is_valid}")
    
    # JSON -> YAML
    converter.json_to_yaml_config(json_config, "output_config.yaml")
    print("\n已保存YAML配置到: output_config.yaml")
```

```typescript
  // Dam break

  h_left?: number;
  h_right?: number;
  Q_left?: number;
  Q_right?: number;
}

// 仿真配置类型
export interface SimulationConfig {

  width: number;
  length: number;
  n_cells: number;
  
  // Physical parameters
  manning_n: number;
  slope: number;
  
  // Numerical parameters
  cfl: number;
  order: 1 | 2;
  use_numba: boolean;
  
  // Time parameters
  t_end: number;
  dt_max: number;
  output_interval: number;
  
  // Conditions
  initial_conditions: InitialCondition;
  boundary_conditions: {
    upstream: BoundaryCondition;

// 性能指标类型
```

```typescript
    order: 2,
    use_numba: true,
    t_end: 10.0,
    dt_max: 0.1,
    output_interval: 0.5,
    initial_conditions: {
      type: 'uniform',
      h: 5.0,
      Q: 10.0
    },
    boundary_conditions: {
      upstream: { type: 'Q', value: 10.0 },
      downstream: { type: 'h', value: 5.0 }
    }
  });
  
  const [result, setResult] = useState<SimulationResult | null>(null);
  const [loading, setLoading] = useState(false);
  
  const runSimulation = async () => {
    setLoading(true);
    try {
      // 提交仿真
      const response = await fetch('http://localhost:8000/api/v1/simulations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          name: 'Test Simulation',
          config: config
        })
      });
      
      const data = await response.json();
      const taskId = data.task_id;
      
      // 轮询等待完成
      const pollResult = async () => {
        const statusResponse = await fetch(
          `http://localhost:8000/api/v1/simulations/${taskId}/status`
        );
        const status = await statusResponse.json();
        
        if (status.status === 'completed') {
          // 获取结果
          const resultResponse = await fetch(
            `http://localhost:8000/api/v1/simulations/${taskId}/results`
          );
          const result = await resultResponse.json();
          setResult(result);
          setLoading(false);
        } else if (status.status === 'failed') {
          console.error('Simulation failed:', status.error);
          setLoading(false);
        } else {
          // 继续轮询
          setTimeout(pollResult, 1000);
        }
      };
      
      pollResult();
    } catch (error) {
      console.error('Error:', error);
      setLoading(false);
    }
  };
  
  // 渲染图表
  const renderChart = () => {
    if (!result) return null;
    
    // 绘制最后时刻的水深分布
    const lastTimeIdx = result.time.length - 1;
    const chartData = {
      labels: result.x,
      datasets: [{
        label: 'Water Depth (m)',
        data: result.h[lastTimeIdx],
        borderColor: 'rgb(75, 192, 192)',
        backgroundColor: 'rgba(75, 192, 192, 0.2)',
      }]
    };
    
    return (
      <div>
        <h3>Results</h3>
        <Line data={chartData} />
        <div>
          <p>Max Depth: {result.metrics.max_depth.toFixed(2)} m</p>
          <p>Mass Error: {result.metrics.mass_conservation_error.toFixed(6)}</p>
          <p>Duration: {result.duration.toFixed(2)} s</p>
        </div>
      </div>
    );
  };
  
  return (
    <div>
      <h2>Simulation Runner</h2>
      <button onClick={runSimulation} disabled={loading}>
        {loading ? 'Running...' : 'Run Simulation'}
      </button>
      {renderChart()}
    </div>
  );
};

export default SimulationRunner;
```

## 📈 统一结果分析模块

### 创建结果分析工具

文件：`/workspace/web/backend/utils/result_analyzer.py`

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""
统一结果分析模块
"""

import numpy as np
from typing import Dict, List, Tuple
from web.backend.core.hydraulic_engine import SimulationResult

class ResultAnalyzer:
    """结果分析器"""
    
    @staticmethod
    def extract_time_series(result: SimulationResult, x_position: float) -> Dict:
        """
        提取指定位置的时间序列
        
        Args:
            result: 仿真结果
            x_position: 空间位置 (m)
            
        Returns:
            时间序列数据字典
        """
        # 找到最接近的网格点
        x_array = np.array(result.x)
        x_idx = np.argmin(np.abs(x_array - x_position))
        
        # 提取时间序列
        time_series = {
            "time": result.time,
            "h": [h_t[x_idx] for h_t in result.h],
            "Q": [Q_t[x_idx] for Q_t in result.Q],
            "V": [V_t[x_idx] for V_t in result.V],
            "position": result.x[x_idx]
        }
        
        return time_series
    
    @staticmethod
    def extract_spatial_profile(result: SimulationResult, time_point: float) -> Dict:
        """
        提取指定时刻的空间分布
        
        Args:
            result: 仿真结果
            time_point: 时间点 (s)
            
        Returns:
            空间分布数据字典
        """
        # 找到最接近的时间步
        time_array = np.array(result.time)
        t_idx = np.argmin(np.abs(time_array - time_point))
        
        # 提取空间分布
        spatial_profile = {
            "x": result.x,
            "h": result.h[t_idx],
            "Q": result.Q[t_idx],
            "V": result.V[t_idx],
            "time": result.time[t_idx]
        }
        
        return spatial_profile
    
    @staticmethod
    def calculate_statistics(result: SimulationResult) -> Dict:
        """
        计算统计指标
        
        Args:
            result: 仿真结果
            
        Returns:
            统计指标字典
        """
        h_array = np.array(result.h)
        Q_array = np.array(result.Q)
        V_array = np.array(result.V)
        
        stats = {
            # 水深统计
            "h_mean": float(np.mean(h_array)),
            "h_std": float(np.std(h_array)),
            "h_max": float(np.max(h_array)),
            "h_min": float(np.min(h_array)),
            
            # 流量统计
            "Q_mean": float(np.mean(Q_array)),
            "Q_std": float(np.std(Q_array)),
            "Q_max": float(np.max(Q_array)),
            "Q_min": float(np.min(Q_array)),
            
            # 流速统计
            "V_mean": float(np.mean(V_array)),
            "V_std": float(np.std(V_array)),
            "V_max": float(np.max(V_array)),
            "V_min": float(np.min(V_array)),
        }
        
        return stats
    
    @staticmethod
    def detect_shock_waves(result: SimulationResult, threshold: float = 0.1) -> List[Tuple]:
        """
        检测水跃/激波位置
        
        Args:
            result: 仿真结果
            threshold: 水深梯度阈值 (m/m)
            
        Returns:
            激波位置列表 [(time, position), ...]
        """
        shocks = []
        
        for t_idx, (h_t, t) in enumerate(zip(result.h, result.time)):
            h_array = np.array(h_t)
            
            # 计算水深梯度
            dh_dx = np.diff(h_array) / np.diff(result.x)
            
            # 找到梯度超过阈值的位置
            shock_indices = np.where(np.abs(dh_dx) > threshold)[0]
            
            for idx in shock_indices:
                shock_position = (result.x[idx] + result.x[idx + 1]) / 2
                shocks.append((t, shock_position))
        
        return shocks

# 使用示例
if __name__ == "__main__":
    from web.backend.core.hydraulic_engine import HydraulicEngine
    
    # 运行仿真
    engine = HydraulicEngine()
    result = engine.run_canal_simulation(
        task_id="analysis_test",
        config={...}
    )
    
    # 分析结果
    analyzer = ResultAnalyzer()
    
    # 提取中点时间序列
    time_series = analyzer.extract_time_series(result, x_position=500.0)
    print("中点水深变化:", time_series["h"][:5])
    
    # 提取最终空间分布
    final_profile = analyzer.extract_spatial_profile(result, time_point=result.time[-1])
    print("最终水深分布:", final_profile["h"][:5])
    
    # 计算统计指标
    stats = analyzer.calculate_statistics(result)
    print("平均水深:", stats["h_mean"])
    
    # 检测激波
    shocks = analyzer.detect_shock_waves(result)
    print(f"检测到 {len(shocks)} 个激波")
```
