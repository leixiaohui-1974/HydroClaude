   ```typescript
   async runSimulation(
     componentId: string,
     config: Record<string, any>
   ): Promise<ApiResponse<SimulationResponse>> {
     const component = getComponentById(componentId);
     
     const response = await this.axiosInstance.post(
       component.apiEndpoint,
       config
     );

   }
   ```

   ```typescript
   async runBatchSimulations(
     simulations: Array<{ componentId: string; config: any }>
   ) {
     const results = await Promise.all(
       simulations.map(sim => this.runSimulation(sim.componentId, sim.config))
     );
     
     return {
       success: true,
       data: results.filter(r => r.success)
     };
   }
   ```

### 4. 完整工作流

```typescript
import { useState } from 'react';
import { UnifiedComponentSelector } from '@/components/UnifiedComponentSelector';
import { ComponentConfigForm } from '@/components/ComponentConfigForm';
import { unifiedComponentApi } from '@/services/unifiedComponentApi';

function SimulationWorkflow() {
  const [selectedComponent, setSelectedComponent] = useState(null);
  const [result, setResult] = useState(null);
  
  const handleSelectComponent = (component) => {
    setSelectedComponent(component);
  };
  
  const handleSubmitConfig = async (values) => {
    const result = await unifiedComponentApi.runSimulation(
      selectedComponent.id,
      values
    );
    
    if (result.success) {
      setResult(result.data);
    }
  };
  
  return (
    <div>
      {/* 步骤1: 选择组件 */}
      {!selectedComponent && (
        <UnifiedComponentSelector
          onSelectComponent={handleSelectComponent}
        />
      )}
      
      {/* 步骤2: 配置参数 */}
      {selectedComponent && !result && (
        <ComponentConfigForm
          component={selectedComponent}
          onSubmit={handleSubmitConfig}
        />
      )}
      
      {/* 步骤3: 查看结果 */}
      {result && (
        <div>
          <h3>仿真结果</h3>
          <pre>{JSON.stringify(result.metrics, null, 2)}</pre>
        </div>
      )}
    </div>
  );
}
```
