## v1.4.0 Feature Design

### Feature 1: Time-Series Animation 🎬

#### Overview
Add automatic playback capability to visualize simulation evolution over time.

#### Components

**1.1 Animation Controller**
```typescript
interface AnimationController {
  isPlaying: boolean;
  playbackSpeed: number; // 1x, 2x, 5x, 10x
  currentFrame: number;
  totalFrames: number;
  loop: boolean;

  play(): void;
  pause(): void;
  stop(): void;
  step(direction: 'forward' | 'backward'): void;
  setSpeed(speed: number): void;
}
```

**1.2 Animation Controls UI**
- ⏯️ Play/Pause button
- ⏹️ Stop button
- ⏮️ Previous frame
- ⏭️ Next frame
- 🔄 Loop toggle
- 🎚️ Speed selector (1x, 2x, 5x, 10x)
- 📊 Frame counter (e.g., "Frame 45/150")

**1.3 Animated Plots**
- Update plots automatically during playback
- Smooth transitions between frames
- Configurable frame rate (10-60 FPS)
- Buffer management for smooth playback

#### Implementation

**File**: `web/frontend/src/features/simulation/components/AnimationController.tsx`

```typescript
import { useState, useEffect, useRef } from 'react';
import { Button, Space, Select, Switch, Typography } from 'antd';
import {
  PlayCircleOutlined,
  PauseCircleOutlined,
  StepBackwardOutlined,
  StepForwardOutlined,
  ReloadOutlined
} from '@ant-design/icons';

interface AnimationControllerProps {
  totalFrames: number;
  currentFrame: number;
  onFrameChange: (frame: number) => void;
}

const AnimationController = ({
  totalFrames,
  currentFrame,
  onFrameChange
}: AnimationControllerProps) => {
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackSpeed, setPlaybackSpeed] = useState(1);
  const [loop, setLoop] = useState(true);
  const intervalRef = useRef<number | null>(null);

  // Auto-play logic
  useEffect(() => {
    if (isPlaying) {
      const fps = 30;
      const interval = (1000 / fps) / playbackSpeed;

      intervalRef.current = window.setInterval(() => {
        onFrameChange((prev) => {
          const next = prev + 1;
          if (next >= totalFrames) {
            if (loop) return 0;
            setIsPlaying(false);
            return prev;
          }
          return next;
        });
      }, interval);
    }

    return () => {
      if (intervalRef.current) clearInterval(intervalRef.current);
    };
  }, [isPlaying, playbackSpeed, loop, totalFrames]);

  const handlePlay = () => setIsPlaying(true);
  const handlePause = () => setIsPlaying(false);
  const handleStop = () => {
    setIsPlaying(false);
    onFrameChange(0);
  };
  const handleStepBackward = () => {
    onFrameChange(Math.max(0, currentFrame - 1));
  };
  const handleStepForward = () => {
    onFrameChange(Math.min(totalFrames - 1, currentFrame + 1));
  };

  return (
    <Space direction="vertical" style={{ width: '100%' }}>
      <Space wrap>
        <Button
          icon={<StepBackwardOutlined />}
          onClick={handleStepBackward}
          disabled={currentFrame === 0}
        >
          上一帧
        </Button>

        {isPlaying ? (
          <Button
            icon={<PauseCircleOutlined />}
            onClick={handlePause}
            type="primary"
          >
            暂停
          </Button>
        ) : (
          <Button
            icon={<PlayCircleOutlined />}
            onClick={handlePlay}
            type="primary"
          >
            播放
          </Button>
        )}

        <Button
          icon={<ReloadOutlined />}
          onClick={handleStop}
        >
          重置
        </Button>

        <Button
          icon={<StepForwardOutlined />}
          onClick={handleStepForward}
          disabled={currentFrame === totalFrames - 1}
        >
          下一帧
        </Button>

        <Select
          value={playbackSpeed}
          onChange={setPlaybackSpeed}
          style={{ width: 100 }}
          options={[
            { label: '0.5x', value: 0.5 },
            { label: '1x', value: 1 },
            { label: '2x', value: 2 },
            { label: '5x', value: 5 },
            { label: '10x', value: 10 }
          ]}
        />

        <Space>
          <Typography.Text>循环播放</Typography.Text>
          <Switch checked={loop} onChange={setLoop} />
        </Space>
      </Space>

      <Typography.Text type="secondary">
        帧 {currentFrame + 1} / {totalFrames}
      </Typography.Text>
    </Space>
  );
};

export default AnimationController;
```

### Feature 3: Real-Time Monitoring Dashboard 📊

#### Overview
Display live updates during simulation execution with WebSocket support.

#### Components

**3.1 WebSocket Connection**
```typescript
interface WebSocketMessage {
  type: 'progress' | 'metrics' | 'snapshot';
  data: {
    progress: number;      // 0-100%
    currentTime: number;   // Physical time
    stepCount: number;     // Iteration count
    metrics?: {
      mass_conservation: number;
      max_depth: number;
      max_velocity: number;

      Q: number[];
      V: number[];

  };
}
```

**3.2 Live Dashboard UI**
- 📊 Progress bar with percentage
- ⏱️ Real-time metrics (updating every second)
- 📈 Live plot of key metrics
- 🎬 Streaming visualization (optional)

#### Implementation

**File**: `web/frontend/src/features/simulation/components/LiveDashboard.tsx`

```typescript
import { useState, useEffect } from 'react';
import { Card, Progress, Row, Col, Statistic } from 'antd';
import Plot from 'react-plotly.js';

interface LiveDashboardProps {
  taskId: string;
  wsUrl: string;
}

const LiveDashboard = ({ taskId, wsUrl }: LiveDashboardProps) => {
  const [progress, setProgress] = useState(0);
  const [metrics, setMetrics] = useState({
    currentTime: 0,
    stepCount: 0,
    massConservation: 0,
    maxDepth: 0,
    maxVelocity: 0
  });
  const [metricsHistory, setMetricsHistory] = useState<any[]>([]);

  useEffect(() => {
    const ws = new WebSocket(`${wsUrl}/${taskId}`);

    ws.onmessage = (event) => {
      const message = JSON.parse(event.data);

      if (message.type === 'progress') {
        setProgress(message.data.progress);
        setMetrics({
          currentTime: message.data.currentTime,
          stepCount: message.data.stepCount,
          massConservation: message.data.metrics?.mass_conservation || 0,
          maxDepth: message.data.metrics?.max_depth || 0,
          maxVelocity: message.data.metrics?.max_velocity || 0
        });

        // Update history for live charts
        setMetricsHistory(prev => [...prev, {
          time: message.data.currentTime,
          mass: message.data.metrics?.mass_conservation || 0,
          depth: message.data.metrics?.max_depth || 0
        }]);
      }
    };

    return () => ws.close();
  }, [taskId, wsUrl]);

  return (
    <Card title="🔴 实时监控" extra={`进度: ${progress.toFixed(1)}%`}>
      <Progress percent={progress} status="active" />

      <Row gutter={16} style={{ marginTop: 20 }}>
        <Col span={6}>
          <Statistic
            title="物理时间"
            value={metrics.currentTime}
            precision={2}
            suffix="s"
          />
        </Col>
        <Col span={6}>
          <Statistic
            title="时间步数"
            value={metrics.stepCount}
          />
        </Col>
        <Col span={6}>
          <Statistic
            title="质量守恒"
            value={metrics.massConservation}
            precision={6}
            valueStyle={{ color: metrics.massConservation < 1e-6 ? '#3f8600' : '#cf1322' }}
          />
        </Col>
        <Col span={6}>
          <Statistic
            title="最大水深"
            value={metrics.maxDepth}
            precision={3}
            suffix="m"
          />
        </Col>
      </Row>

      {metricsHistory.length > 10 && (
        <div style={{ marginTop: 20 }}>
          <Plot
            data={[
              {
                x: metricsHistory.map(m => m.time),
                y: metricsHistory.map(m => m.mass),
                type: 'scatter',
                mode: 'lines',
                name: '质量守恒误差',
                line: { color: '#1890ff' }
              }
            ]}
            layout={{
              title: '实时质量守恒监控',
              xaxis: { title: '时间 (s)' },
              yaxis: { title: '误差', type: 'log' },
              height: 200,
              margin: { t: 40, r: 20, b: 40, l: 60 }
            }}
            config={{ responsive: true }}
            style={{ width: '100%' }}
          />
        </div>
      )}
    </Card>
  );
};

export default LiveDashboard;
```

#### Backend Support (FastAPI)

**File**: `web/backend/websocket.py` (New)

```python
from fastapi import WebSocket, WebSocketDisconnect
from typing import Dict, Set
import asyncio
import json

class ConnectionManager:
    def __init__(self):
        self.active_connections: Dict[str, Set[WebSocket]] = {}

    async def connect(self, task_id: str, websocket: WebSocket):
        await websocket.accept()
        if task_id not in self.active_connections:
            self.active_connections[task_id] = set()
        self.active_connections[task_id].add(websocket)

    def disconnect(self, task_id: str, websocket: WebSocket):
        if task_id in self.active_connections:
            self.active_connections[task_id].remove(websocket)
            if not self.active_connections[task_id]:
                del self.active_connections[task_id]

    async def broadcast(self, task_id: str, message: dict):
        if task_id in self.active_connections:
            disconnected = set()
            for connection in self.active_connections[task_id]:
                try:
                    await connection.send_json(message)
                except:
                    disconnected.add(connection)

            # Clean up disconnected
            for conn in disconnected:
                self.disconnect(task_id, conn)

manager = ConnectionManager()

@app.websocket("/ws/simulation/{task_id}")
async def websocket_endpoint(websocket: WebSocket, task_id: str):
    await manager.connect(task_id, websocket)
    try:
        while True:
            await websocket.receive_text()  # Keep alive
    except WebSocketDisconnect:
        manager.disconnect(task_id, websocket)

# In simulation execution loop:
async def run_simulation_with_progress(task_id: str, config: dict):
    # ... simulation setup ...

    for step in range(total_steps):
        # ... solve one step ...

        # Send progress update every 10 steps
        if step % 10 == 0:
            progress_pct = (step / total_steps) * 100
            await manager.broadcast(task_id, {
                "type": "progress",
                "data": {
                    "progress": progress_pct,
                    "currentTime": solver.t,
                    "stepCount": step,
                    "metrics": {
                        "mass_conservation": solver.get_mass_conservation_error(),
                        "max_depth": solver.h.max(),
                        "max_velocity": solver.V.max()
                    }
                }
            })

    # ... finalize ...
```

```typescript
import Plot from 'react-plotly.js';
import { Card, Tabs } from 'antd';

interface EnhancedChartsProps {
  x: number[];
  time: number[];
  h: number[][];
  V: number[][];
  Q: number[][];
}

const EnhancedCharts = ({ x, time, h, V, Q }: EnhancedChartsProps) => {
  // Contour plot
  const contourPlot = {
    data: [{
      type: 'contour',
      x: x,
      y: time,
      z: h,
      colorscale: 'Jet',
      colorbar: { title: '水深 (m)' },
      contours: {
        showlabels: true,
        labelfont: { size: 12, color: 'white' }
      }
    }],
    layout: {
      title: '水深等值线图',
      xaxis: { title: '位置 (m)' },
      yaxis: { title: '时间 (s)' },
      height: 400
    }
  };

  // Heatmap
  const heatmapPlot = {
    data: [{
      type: 'heatmap',
      x: x,
      y: time,
      z: V,
      colorscale: 'RdBu',
      colorbar: { title: '流速 (m/s)' }
    }],
    layout: {
      title: '流速热力图',
      xaxis: { title: '位置 (m)' },
      yaxis: { title: '时间 (s)' },
      height: 400
    }
  };

  // Time series at specific location
  const midpoint = Math.floor(x.length / 2);
  const timeSeriesPlot = {
    data: [
      {
        x: time,
        y: h.map(frame => frame[midpoint]),
        type: 'scatter',
        mode: 'lines',
        name: '水深',
        line: { color: '#1890ff' }
      },
      {
        x: time,
        y: V.map(frame => frame[midpoint]),
        type: 'scatter',
        mode: 'lines',
        name: '流速',
        yaxis: 'y2',
        line: { color: '#52c41a' }
      }
    ],
    layout: {
      title: `中点时间序列 (x = ${x[midpoint].toFixed(1)}m)`,
      xaxis: { title: '时间 (s)' },
      yaxis: { title: '水深 (m)', side: 'left' },
      yaxis2: {
        title: '流速 (m/s)',
        overlaying: 'y',
        side: 'right'
      },
      height: 400
    }
  };

  return (
    <Tabs
      items={[
        {
          key: 'contour',
          label: '等值线图',
          children: (
            <Plot
              data={contourPlot.data as any}
              layout={contourPlot.layout as any}
              config={{ responsive: true }}
              style={{ width: '100%' }}
            />
          )
        },
        {
          key: 'heatmap',
          label: '热力图',
          children: (
            <Plot
              data={heatmapPlot.data as any}
              layout={heatmapPlot.layout as any}
              config={{ responsive: true }}
              style={{ width: '100%' }}
            />
          )
        },
        {
          key: 'timeseries',
          label: '时间序列',
          children: (
            <Plot
              data={timeSeriesPlot.data as any}
              layout={timeSeriesPlot.layout as any}
              config={{ responsive: true }}
              style={{ width: '100%' }}
            />
          )
        }
      ]}

  );
};
```
