【JavaScript - 浏览器控制台】
fetch('http://localhost:8000/api/structures/pump', {
  method: 'POST',
  headers: {'Content-Type': 'application/json'},
  body: JSON.stringify({
    pump: {flow_rate: 10, head: 15, num_pumps: 2, pump_type: 'parallel'},
    upstream: {water_level: 5.0},
    downstream: {elevation: 20.0},
    operation: {duration: 100}
  })
})
.then(r => r.json())
.then(d => console.log('流量:', d.metrics.avg_flow, 'm³/s'));

【Python - 命令行】
import requests
r = requests.post('http://localhost:8000/api/structures/pump', json={
    'pump': {'flow_rate': 10, 'head': 15, 'num_pumps': 2, 'pump_type': 'parallel'},
    'upstream': {'water_level': 5.0},
    'downstream': {'elevation': 20.0},
    'operation': {'duration': 100}
})
print('效率:', r.json()['metrics']['avg_efficiency'] * 100, '%')

【cURL - 命令行】
curl -X POST http://localhost:8000/api/structures/pump \
  -H 'Content-Type: application/json' \
  -d '{"pump":{"flow_rate":10,"head":15,"num_pumps":2,"pump_type":"parallel"},"upstream":{"water_level":5},"downstream":{"elevation":20},"operation":{"duration":100}}'
