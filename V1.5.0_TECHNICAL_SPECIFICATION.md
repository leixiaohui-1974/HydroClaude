### Architecture Principles for v1.5.0

1. **Client-Side First**: All new features are client-side only
2. **No Backend Changes**: Backend remains unchanged (simulation only)
3. **Browser Storage**: Use localStorage + File API for persistence
4. **Progressive Enhancement**: Features degrade gracefully
5. **Performance**: Minimize re-renders, optimize large datasets

## Data Models

### 1. HydraulicModel

**Purpose**: Represents a complete hydraulic model with all configuration

**TypeScript Definition**:
```typescript
// types/model.ts

export interface HydraulicModel {
  // Metadata
  id: string;                    // UUID v4
  name: string;                  // User-friendly name
  description?: string;          // Optional description
  author?: string;               // Creator name
  createdAt: string;             // ISO 8601 datetime
  updatedAt: string;             // ISO 8601 datetime
  version: string;               // Format version (e.g., "1.5.0")
  tags: string[];                // Categorization tags

  // Model Configuration
  nodes: FlowchartNode[];        // ReactFlow nodes
  edges: FlowchartEdge[];        // ReactFlow edges

  globalConfig: GlobalConfig;    // Simulation parameters

  // Validation
  isValid: boolean;              // Validation status
  validationErrors: string[];    // Error messages

  // Optional UI State
  viewport?: {                   // ReactFlow viewport
    x: number;
    y: number;
    zoom: number;
  };

  // Optional Preview
  thumbnail?: string;            // base64 encoded PNG (max 100KB)
}

export interface GlobalConfig {
  gravity: number;               // m/s² (default: 9.81)
  roughness: number;             // Manning's n (default: 0.03)
  timeStep: number;              // seconds (default: 0.01)
  duration: number;              // seconds (default: 10.0)
  saveInterval: number;          // seconds (default: 0.1)
  solver?: {
    type: 'godunov' | 'roe';    // Riemann solver
    limiter?: 'minmod' | 'superbee' | 'vanleer';
    cfl?: number;               // CFL number (default: 0.9)
  };
}

export interface FlowchartNode {
  id: string;
  type: 'channel' | 'boundary' | 'junction';
  position: { x: number; y: number };
  data: NodeData;
}

export interface NodeData {
  // Channel-specific
  length?: number;              // meters
  width?: number;               // meters
  manning?: number;             // roughness
  slope?: number;               // dimensionless
  initialDepth?: InitialDepth;

  // Boundary-specific
  type?: 'wall' | 'fixedDepth' | 'fixedDischarge';
  depth?: number;               // for fixedDepth
  discharge?: number;           // for fixedDischarge
  timeSeries?: { t: number; value: number }[];

  // Junction-specific
  junctionType?: 'merge' | 'split';
}

export interface InitialDepth {
  type: 'constant' | 'step' | 'linear' | 'custom';
  depth?: number;               // for constant
  leftDepth?: number;           // for step
  rightDepth?: number;          // for step
  position?: number;            // for step
  values?: number[];            // for custom
}

export interface FlowchartEdge {
  id: string;
  source: string;               // node id
  target: string;               // node id
  type?: string;                // ReactFlow edge type
}
```

**Storage Format** (JSON):
```json
{
  "format": "HydroClaude Model",
  "version": "1.5.0",
  "model": { ... }
}
```

**Size Limits**:
- Max model size: 5 MB
- Max nodes: 1,000
- Max edges: 2,000
- Max thumbnail: 100 KB

### Middleware

#### localStorage Sync Middleware

```typescript
// store/middleware/storageMiddleware.ts

export const storageMiddleware: Middleware = store => next => action => {
  const result = next(action);

  // Sync specific actions to localStorage
  if (action.type.startsWith('modeling/')) {
    const state = store.getState();
    localStorage.setItem(
      STORAGE_KEYS.MODELS,
      JSON.stringify(state.modeling.savedModels)
    );
  }

  if (action.type.startsWith('comparison/')) {
    const state = store.getState();
    localStorage.setItem(
      STORAGE_KEYS.COMPARISONS,
      JSON.stringify(state.comparison.sessions)
    );
  }

  return result;
};
```

## Storage Strategy

### localStorage Usage

**Capacity Planning**:
- localStorage limit: ~5-10 MB (browser dependent)
- Target usage: < 5 MB total
- Per-model limit: 500 KB
- Max models stored: ~10 models

**Quota Management**:
```typescript
// utils/storage.ts

export const checkStorageQuota = (): {
  used: number;
  available: number;
  percentage: number;
} => {
  let used = 0;
  for (const key in localStorage) {
    if (key.startsWith('hydroclaude_')) {
      used += localStorage[key].length;
    }
  }

  const available = 5 * 1024 * 1024; // 5 MB conservative limit
  const percentage = (used / available) * 100;

  return { used, available, percentage };
};

export const cleanupOldModels = (keepCount: number = 10) => {
  const models = JSON.parse(localStorage.getItem(STORAGE_KEYS.MODELS) || '[]');

  // Sort by last accessed, keep most recent
  const sorted = models.sort((a, b) =>
    new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime()
  );

  const keep = sorted.slice(0, keepCount);
  localStorage.setItem(STORAGE_KEYS.MODELS, JSON.stringify(keep));
};
```

**Data Compression** (Optional):
```typescript
// Use LZ-string for compression if needed
import LZString from 'lz-string';

export const saveCompressed = (key: string, data: any) => {
  const json = JSON.stringify(data);
  const compressed = LZString.compress(json);
  localStorage.setItem(key, compressed);
};

export const loadCompressed = (key: string): any => {
  const compressed = localStorage.getItem(key);
  if (!compressed) return null;
  const json = LZString.decompress(compressed);
  return JSON.parse(json);
};
```

```typescript
export const exportModelJSON = (model: HydraulicModel): void => {
  const exportData = {
    format: 'HydroClaude Model',
```

#### CSV Export

```typescript
export const exportModelCSV = (model: HydraulicModel): void => {
  // Node data
  const nodeHeaders = ['id', 'type', 'x', 'y', 'length', 'width', 'manning'];
  const nodeRows = model.nodes.map(node => [
    node.id,
    node.type,
    node.position.x,
    node.position.y,
    node.data.length || '',
    node.data.width || '',
    node.data.manning || ''
  ]);

  const csv = [
    nodeHeaders.join(','),
    ...nodeRows.map(row => row.join(','))
  ].join('\n');

  const blob = new Blob([csv], { type: 'text/csv' });
  const filename = `${model.name}_nodes.csv`;

  downloadBlob(blob, filename);
};
```

#### Excel Export (Results)

```typescript
export const exportResultsExcel = (
  result: SimulationResult,
  modelName: string
): void => {
  const wb = XLSX.utils.book_new();

  // Sheet 1: Summary
  const summaryData = [
    ['Model Name', modelName],
    ['Simulation Date', result.metadata.simulationDate],
    ['Time Steps', result.time.length],
    ['Spatial Points', result.x.length],
    ['', ''],
    ['Metrics', ''],
    ['Max Depth', result.metrics.max_depth],
    ['Max Velocity', result.metrics.max_velocity],
    ['Mass Error', result.metrics.mass_conservation_error]
  ];
  const ws1 = XLSX.utils.aoa_to_sheet(summaryData);
  XLSX.utils.book_append_sheet(wb, ws1, 'Summary');

  // Sheet 2: Time series
  const timeSeriesData = [
    ['Time (s)', 'Max Depth (m)', 'Max Velocity (m/s)']
  ];
  for (let t = 0; t < result.time.length; t++) {
    timeSeriesData.push([
      result.time[t],
      Math.max(...result.h[t]),
      Math.max(...result.V[t])
    ]);
  }
  const ws2 = XLSX.utils.aoa_to_sheet(timeSeriesData);
  XLSX.utils.book_append_sheet(wb, ws2, 'Time Series');

  // Sheet 3: Spatial (final time)
  const finalIdx = result.time.length - 1;
  const spatialData = [['Position (m)', 'Depth (m)', 'Velocity (m/s)', 'Discharge (m³/s)']];
  for (let i = 0; i < result.x.length; i++) {
    spatialData.push([
      result.x[i],
      result.h[finalIdx][i],
      result.V[finalIdx][i],
      result.Q[finalIdx][i]
    ]);
  }
  const ws3 = XLSX.utils.aoa_to_sheet(spatialData);
  XLSX.utils.book_append_sheet(wb, ws3, 'Spatial Distribution');

  // Write file
  XLSX.writeFile(wb, `${modelName}_results.xlsx`);
};
```

#### JSON Import

```typescript
export const importModelJSON = async (file: File): Promise<HydraulicModel> => {
  try {
    const text = await file.text();
    const data = JSON.parse(text);

    // Validate format
    if (data.format !== 'HydroClaude Model') {
      throw new Error('Invalid file format. Expected HydroClaude Model.');
    }

    // Version check
    if (!data.version || !isCompatibleVersion(data.version)) {
      console.warn(`Model version ${data.version} may not be fully compatible`);
    }

    // Validate model structure
    const model = data.model as HydraulicModel;
    validateModel(model);

    return model;
  } catch (error) {
    throw new Error(`Import failed: ${error.message}`);
  }
};

const isCompatibleVersion = (version: string): boolean => {
  const [major] = version.split('.').map(Number);
  return major === 1; // v1.x.x compatible
};

const validateModel = (model: HydraulicModel): void => {
  if (!model.id || !model.name) {
    throw new Error('Invalid model: missing required fields');
  }

  if (!Array.isArray(model.nodes) || !Array.isArray(model.edges)) {
    throw new Error('Invalid model: nodes and edges must be arrays');
  }

  if (!model.globalConfig) {
    throw new Error('Invalid model: missing globalConfig');
  }
};
```

```typescript
// Validate all imported data
const validateModelInput = (model: any): ValidationResult => {
  const errors: string[] = [];

  // Check required fields
  if (!model.id || typeof model.id !== 'string') {
    errors.push('Invalid model ID');
  }

  // Check data types
  if (!Array.isArray(model.nodes)) {
    errors.push('Nodes must be an array');
  }

  // Check ranges
  if (model.nodes.length > 1000) {
    errors.push('Too many nodes (max 1000)');
  }

  // Sanitize strings
  model.name = sanitizeString(model.name);
  model.description = sanitizeString(model.description);

  return { valid: errors.length === 0, errors };
};
```
