## v1.5.0 Feature Design

### Feature 1: Model Import/Export 📦

#### Overview
Enable users to save, load, and share hydraulic models in multiple formats.

#### 1.1 Model Data Structure

```typescript
// types/model.ts
export interface HydraulicModel {
  // Metadata
  id: string;
  name: string;
  description: string;
  author?: string;
  createdAt: string;
  updatedAt: string;
  version: string;
  tags: string[];

  // Model configuration
  nodes: FlowchartNode[];
  edges: FlowchartEdge[];
  globalConfig: {
    gravity: number;
    roughness: number;
    timeStep: number;
    duration: number;
    saveInterval: number;
  };

  // Validation status
  isValid: boolean;
  validationErrors: string[];

  // Optional preview
  thumbnail?: string; // base64 encoded image
}
```

#### 1.2 Export Formats

**JSON Format** (.hydro.json)
```json
{
  "format": "HydroClaude Model",
  "version": "1.5.0",
  "model": {
    "id": "model-123",
    "name": "Dam Break Simulation",
    "description": "Classical dam break test case",
    "nodes": [...],
    "edges": [...],
    "globalConfig": {...}
  }
}
```

**Legacy Format Support** (.xml, .inp)
- Import from HEC-RAS (.xml)
- Import from SWMM (.inp)
- Convert to HydroClaude format
- Preserve metadata when possible

#### 1.3 Implementation

**File**: `web/frontend/src/features/modeling/components/ModelIO.tsx`

```typescript
import { Button, Space, message, Upload, Modal } from 'antd';
import { DownloadOutlined, UploadOutlined, SaveOutlined } from '@ant-design/icons';
import { useAppSelector, useAppDispatch } from '../../../store/hooks';
import { saveModel, loadModel } from '../modelingSlice';
import type { HydraulicModel } from '../../../types/model';

const ModelIO = () => {
  const dispatch = useAppDispatch();
  const currentModel = useAppSelector(state => state.modeling);

  // Export model as JSON
  const handleExportJSON = () => {
    const model: HydraulicModel = {
      id: crypto.randomUUID(),
      name: currentModel.name || 'Untitled Model',
      description: currentModel.description || '',
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      version: '1.5.0',
      tags: currentModel.tags || [],
      nodes: currentModel.nodes,
      edges: currentModel.edges,
      globalConfig: currentModel.config,
      isValid: currentModel.isValid,
      validationErrors: currentModel.validationErrors
    };

    const json = JSON.stringify(model, null, 2);
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${model.name.replace(/\s+/g, '_')}.hydro.json`;
    a.click();
    URL.revokeObjectURL(url);

    message.success('模型已导出');
  };

  // Export as CSV (simplified format)
  const handleExportCSV = () => {
    // Convert nodes to CSV format
    const headers = ['id', 'type', 'position_x', 'position_y', 'length', 'width', 'manning'];
    const rows = currentModel.nodes.map(node => [
      node.id,
      node.type,
      node.position.x,
      node.position.y,
      node.data.length || '',
      node.data.width || '',
      node.data.manning || ''
    ]);

    const csv = [
      headers.join(','),
      ...rows.map(row => row.join(','))
    ].join('\n');

    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${currentModel.name || 'model'}_nodes.csv`;
    a.click();
    URL.revokeObjectURL(url);

    message.success('CSV已导出');
  };

  // Import model from JSON
  const handleImportJSON = async (file: File) => {
    try {
      const text = await file.text();
      const data = JSON.parse(text);

      // Validate format
      if (data.format !== 'HydroClaude Model') {
        throw new Error('Invalid file format');
      }

      // Load model into state
      dispatch(loadModel(data.model));
      message.success(`已导入模型: ${data.model.name}`);
    } catch (error) {
      message.error('导入失败: ' + (error as Error).message);
    }
    return false; // Prevent auto upload
  };

  // Save model to local storage
  const handleSaveLocal = () => {
    const model: HydraulicModel = {
      id: currentModel.id || crypto.randomUUID(),
      name: currentModel.name || 'Untitled Model',
      description: currentModel.description || '',
      createdAt: currentModel.createdAt || new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      version: '1.5.0',
      tags: currentModel.tags || [],
      nodes: currentModel.nodes,
      edges: currentModel.edges,
      globalConfig: currentModel.config,
      isValid: currentModel.isValid,
      validationErrors: currentModel.validationErrors
    };

    // Save to localStorage
    const saved = JSON.parse(localStorage.getItem('hydroclaude_models') || '[]');
    const existing = saved.findIndex((m: HydraulicModel) => m.id === model.id);
    if (existing >= 0) {
      saved[existing] = model;
    } else {
      saved.push(model);
    }
    localStorage.setItem('hydroclaude_models', JSON.stringify(saved));

    // Update Redux state
    dispatch(saveModel(model));
    message.success('模型已保存');
  };

  return (
    <Space>
      <Button
        icon={<SaveOutlined />}
        onClick={handleSaveLocal}
        type="primary"
      >
        保存模型
      </Button>

      <Button
        icon={<DownloadOutlined />}
        onClick={handleExportJSON}
      >
        导出JSON
      </Button>

      <Button
        icon={<DownloadOutlined />}
        onClick={handleExportCSV}
      >
        导出CSV
      </Button>

      <Upload
        beforeUpload={handleImportJSON}
        accept=".json,.hydro.json"
        showUploadList={false}
      >
        <Button icon={<UploadOutlined />}>
          导入模型
        </Button>
      </Upload>
    </Space>
  );
};

export default ModelIO;
```

#### 1.4 Model Library Component

**File**: `web/frontend/src/features/modeling/components/ModelLibrary.tsx`

```typescript
import { useState, useEffect } from 'react';
import { Modal, List, Card, Button, Space, Tag, Input, Popconfirm } from 'antd';
import { DeleteOutlined, CopyOutlined, FolderOpenOutlined } from '@ant-design/icons';
import type { HydraulicModel } from '../../../types/model';

interface ModelLibraryProps {
  visible: boolean;
  onClose: () => void;
  onLoadModel: (model: HydraulicModel) => void;
}

const ModelLibrary = ({ visible, onClose, onLoadModel }: ModelLibraryProps) => {
  const [models, setModels] = useState<HydraulicModel[]>([]);
  const [searchText, setSearchText] = useState('');

  useEffect(() => {
    if (visible) {
      loadModels();
    }
  }, [visible]);

  const loadModels = () => {
    const saved = JSON.parse(localStorage.getItem('hydroclaude_models') || '[]');
    setModels(saved);
  };

  const handleDelete = (id: string) => {
    const filtered = models.filter(m => m.id !== id);
    localStorage.setItem('hydroclaude_models', JSON.stringify(filtered));
    setModels(filtered);
  };

  const handleClone = (model: HydraulicModel) => {
    const cloned: HydraulicModel = {
      ...model,
      id: crypto.randomUUID(),
      name: `${model.name} (副本)`,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    };
    const saved = [...models, cloned];
    localStorage.setItem('hydroclaude_models', JSON.stringify(saved));
    setModels(saved);
  };

  const filteredModels = models.filter(m =>
    m.name.toLowerCase().includes(searchText.toLowerCase()) ||
    m.description?.toLowerCase().includes(searchText.toLowerCase()) ||
    m.tags.some(tag => tag.toLowerCase().includes(searchText.toLowerCase()))
  );

  return (
    <Modal
      title="模型库"
      open={visible}
      onCancel={onClose}
      width={800}
      footer={null}
    >
      <Space direction="vertical" style={{ width: '100%' }}>
        <Input.Search
          placeholder="搜索模型..."
          value={searchText}
          onChange={e => setSearchText(e.target.value)}
          allowClear
        />

        <List
          dataSource={filteredModels}
          renderItem={(model) => (
            <List.Item
              actions={[
                <Button
                  icon={<FolderOpenOutlined />}
                  onClick={() => {
                    onLoadModel(model);
                    onClose();
                  }}
                >
                  打开
                </Button>,
                <Button
                  icon={<CopyOutlined />}
                  onClick={() => handleClone(model)}
                >
                  克隆
                </Button>,
                <Popconfirm
                  title="确定删除此模型？"
                  onConfirm={() => handleDelete(model.id)}
                >
                  <Button icon={<DeleteOutlined />} danger>
                    删除
                  </Button>
                </Popconfirm>
              ]}
            >
              <List.Item.Meta
                title={model.name}
                description={
                  <Space direction="vertical" size="small">
                    <div>{model.description}</div>
                    <div>
                      <small>
                        创建: {new Date(model.createdAt).toLocaleDateString()} |
                        更新: {new Date(model.updatedAt).toLocaleDateString()}
                      </small>
                    </div>
                    <Space>
                      {model.tags.map(tag => (
                        <Tag key={tag} color="blue">{tag}</Tag>
                      ))}
                    </Space>
                  </Space>
                }
              />
            </List.Item>
          )}
        />
      </Space>
    </Modal>
  );
};

export default ModelLibrary;
```

#### 2.2 Implementation

**File**: `web/frontend/src/features/simulation/components/ResultExport.tsx`

```typescript
import { Button, Dropdown, Space, message } from 'antd';
import { DownloadOutlined } from '@ant-design/icons';
import * as XLSX from 'xlsx';
import type { SimulationResult } from '../../../types/simulation';

interface ResultExportProps {
  result: SimulationResult;
  modelName: string;
}

const ResultExport = ({ result, modelName }: ResultExportProps) => {
  // Export as CSV
  const exportCSV = () => {
    const { time, x, h, V, Q } = result;
    const rows: string[] = ['time,position,depth,velocity,discharge'];

    // Flatten time-series data
    for (let t = 0; t < time.length; t++) {
      for (let i = 0; i < x.length; i++) {
        rows.push(
          `${time[t].toFixed(3)},${x[i].toFixed(2)},${h[t][i].toFixed(4)},` +
          `${V[t][i].toFixed(4)},${Q[t][i].toFixed(4)}`
        );
      }
    }

    const csv = rows.join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${modelName}_results.csv`;
    a.click();
    URL.revokeObjectURL(url);

    message.success('CSV已导出');
  };

  // Export as Excel
  const exportExcel = () => {
    const { time, x, h, V, Q, metrics } = result;

    // Create workbook
    const wb = XLSX.utils.book_new();

    // Sheet 1: Summary
    const summaryData = [
      ['模型名称', modelName],
      ['仿真时间', new Date().toLocaleString()],
      ['时间步数', time.length],
      ['空间点数', x.length],
      ['', ''],
      ['性能指标', ''],
      ['质量守恒误差', metrics.mass_conservation_error],
      ['最大水深', metrics.max_depth],
      ['最大流速', metrics.max_velocity],
      ['最大流量', metrics.max_discharge]
    ];
    const ws1 = XLSX.utils.aoa_to_sheet(summaryData);
    XLSX.utils.book_append_sheet(wb, ws1, '摘要');

    // Sheet 2: Time series at specific locations
    const midpoint = Math.floor(x.length / 2);
    const tsData = [
      ['时间 (s)', `水深@${x[midpoint].toFixed(1)}m`, `流速@${x[midpoint].toFixed(1)}m`]
    ];
    for (let t = 0; t < time.length; t++) {
      tsData.push([
        time[t].toFixed(3),
        h[t][midpoint].toFixed(4),
        V[t][midpoint].toFixed(4)
      ]);
    }
    const ws2 = XLSX.utils.aoa_to_sheet(tsData);
    XLSX.utils.book_append_sheet(wb, ws2, '时间序列');

    // Sheet 3: Spatial distribution at final time
    const finalTime = time.length - 1;
    const spatialData = [['位置 (m)', '水深 (m)', '流速 (m/s)', '流量 (m³/s)']];
    for (let i = 0; i < x.length; i++) {
      spatialData.push([
        x[i].toFixed(2),
        h[finalTime][i].toFixed(4),
        V[finalTime][i].toFixed(4),
        Q[finalTime][i].toFixed(4)
      ]);
    }
    const ws3 = XLSX.utils.aoa_to_sheet(spatialData);
    XLSX.utils.book_append_sheet(wb, ws3, '空间分布');

    // Write file
    XLSX.writeFile(wb, `${modelName}_results.xlsx`);
    message.success('Excel已导出');
  };

  // Export as JSON
  const exportJSON = () => {
    const exportData = {
      metadata: {
        model_name: modelName,
        simulation_date: new Date().toISOString(),
        solver: 'Saint-Venant 1D',
        version: '1.5.0'
      },
      configuration: result.config,
      results: {
        time: result.time,
        x: result.x,
        h: result.h,
        V: result.V,
        Q: result.Q
      },
      metrics: result.metrics
    };

    const json = JSON.stringify(exportData, null, 2);
    const blob = new Blob([json], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${modelName}_results.json`;
    a.click();
    URL.revokeObjectURL(url);

    message.success('JSON已导出');
  };

  const menuItems = [
    {
      key: 'csv',
      label: 'CSV格式',
      onClick: exportCSV
    },
    {
      key: 'excel',
      label: 'Excel格式',
      onClick: exportExcel
    },
    {
      key: 'json',
      label: 'JSON格式',
      onClick: exportJSON
    }
  ];

  return (
    <Dropdown menu={{ items: menuItems }} placement="bottomRight">
      <Button icon={<DownloadOutlined />}>
        导出结果
      </Button>
    </Dropdown>
  );
};

export default ResultExport;
```

#### 3.2 Implementation

**File**: `web/frontend/src/features/simulation/components/ComparisonView.tsx`

```typescript
import { useState } from 'react';
import { Card, Row, Col, Slider, Space, Statistic, Select, Button } from 'antd';
import Plot from 'react-plotly.js';
import type { SimulationResult } from '../../../types/simulation';

interface Scenario {
  id: string;
  name: string;
  result: SimulationResult;
  color: string;
}

interface ComparisonViewProps {
  scenarios: Scenario[];
  onRemoveScenario: (id: string) => void;
}

const ComparisonView = ({ scenarios, onRemoveScenario }: ComparisonViewProps) => {
  const [currentFrame, setCurrentFrame] = useState(0);
  const [comparisonMode, setComparisonMode] = useState<'overlay' | 'sidebyside'>('overlay');

  // Get synchronized time range
  const maxFrames = Math.min(...scenarios.map(s => s.result.time.length));
  const currentTime = scenarios[0].result.time[currentFrame];

  // Calculate comparison metrics
  const calculateMetrics = () => {
    if (scenarios.length < 2) return null;

    const ref = scenarios[0].result;
    const comp = scenarios[1].result;

    const maxDepthDiff = Math.abs(
      Math.max(...ref.h[currentFrame]) - Math.max(...comp.h[currentFrame])
    );

    const peakTimeDiff = Math.abs(
      ref.time[ref.h.findIndex(frame => frame.includes(Math.max(...ref.h.flat())))] -
      comp.time[comp.h.findIndex(frame => frame.includes(Math.max(...comp.h.flat())))]
    );

    return {
      maxDepthDiff: maxDepthDiff.toFixed(3),
      peakTimeDiff: peakTimeDiff.toFixed(2)
    };
  };

  const metrics = calculateMetrics();

  // Overlay plot (all scenarios on one plot)
  const overlayPlot = {
    data: scenarios.map(scenario => ({
      x: scenario.result.x,
      y: scenario.result.h[currentFrame],
      type: 'scatter',
      mode: 'lines',
      name: scenario.name,
      line: { color: scenario.color, width: 2 }
    })),
    layout: {
      title: `对比视图 - t = ${currentTime.toFixed(2)}s`,
      xaxis: { title: '位置 (m)' },
      yaxis: { title: '水深 (m)' },
      height: 400,
      legend: { orientation: 'h', y: -0.2 }
    }
  };

  // Side-by-side plots
  const renderSideBySide = () => (
    <Row gutter={16}>
      {scenarios.map(scenario => (
        <Col span={24 / scenarios.length} key={scenario.id}>
          <Card
            title={scenario.name}
            size="small"
            extra={
              <Button
                size="small"
                danger
                onClick={() => onRemoveScenario(scenario.id)}
              >
                移除
              </Button>
            }
          >
            <Plot
              data={[{
                x: scenario.result.x,
                y: scenario.result.h[currentFrame],
                type: 'scatter',
                mode: 'lines',
                line: { color: scenario.color, width: 2 }
              }]}
              layout={{
                xaxis: { title: '位置 (m)' },
                yaxis: { title: '水深 (m)' },
                height: 300,
                margin: { t: 20, r: 20, b: 40, l: 60 }
              }}
              config={{ responsive: true }}
              style={{ width: '100%' }}
            />
          </Card>
        </Col>
      ))}
    </Row>
  );

  return (
    <Space direction="vertical" style={{ width: '100%' }} size="large">
      {/* Scenario summary cards */}
      <Row gutter={16}>
        {scenarios.map(scenario => (
          <Col span={24 / scenarios.length} key={scenario.id}>
            <Card size="small" style={{ borderLeft: `4px solid ${scenario.color}` }}>
              <Statistic
                title={scenario.name}
                value={Math.max(...scenario.result.h[currentFrame])}
                precision={3}
                suffix="m"
                valueStyle={{ fontSize: 16 }}
              />
            </Card>
          </Col>
        ))}
      </Row>

      {/* Time slider */}
      <Card size="small">
        <Space direction="vertical" style={{ width: '100%' }}>
          <div>时间: {currentTime.toFixed(2)} s</div>
          <Slider
            min={0}
            max={maxFrames - 1}
            value={currentFrame}
            onChange={setCurrentFrame}
            marks={{
              0: '0s',
              [maxFrames - 1]: `${scenarios[0].result.time[maxFrames - 1].toFixed(1)}s`
            }}
          />
        </Space>
      </Card>

      {/* Comparison mode selector */}
      <Select
        value={comparisonMode}
        onChange={setComparisonMode}
        style={{ width: 200 }}
        options={[
          { label: '叠加对比', value: 'overlay' },
          { label: '并排对比', value: 'sidebyside' }
        ]}
      />

      {/* Comparison plots */}
      {comparisonMode === 'overlay' ? (
        <Card>
          <Plot
            data={overlayPlot.data as any}
            layout={overlayPlot.layout as any}
            config={{ responsive: true }}
            style={{ width: '100%' }}
          />
        </Card>
      ) : (
        renderSideBySide()
      )}

      {/* Comparison metrics */}
      {metrics && (
        <Card title="对比指标" size="small">
          <Row gutter={16}>
            <Col span={12}>
              <Statistic
                title="最大水深差异"
                value={metrics.maxDepthDiff}
                suffix="m"
              />
            </Col>
            <Col span={12}>
              <Statistic
                title="峰值时间差异"
                value={metrics.peakTimeDiff}
                suffix="s"
              />
            </Col>
          </Row>
        </Card>
      )}
    </Space>
  );
};

export default ComparisonView;
```

```typescript
// useKeyboardShortcuts.ts
import { useEffect } from 'react';

export const useKeyboardShortcuts = (handlers: {
  onNew?: () => void;
  onOpen?: () => void;
  onSave?: () => void;
  onRun?: () => void;
  onUndo?: () => void;
  onRedo?: () => void;
}) => {
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.ctrlKey || e.metaKey) {
        switch (e.key) {
          case 'n':
            e.preventDefault();
            handlers.onNew?.();
            break;
          case 'o':
            e.preventDefault();
            handlers.onOpen?.();
            break;
          case 's':
            e.preventDefault();
            handlers.onSave?.();
            break;
          case 'r':
            e.preventDefault();
            handlers.onRun?.();
            break;
          case 'z':
            e.preventDefault();
            if (e.shiftKey) {
              handlers.onRedo?.();
            } else {
              handlers.onUndo?.();
            }
            break;
        }
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [handlers]);
};
```
